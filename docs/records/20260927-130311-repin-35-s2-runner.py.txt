def _price(secs):
    return secs * (CPU * COST_PER_CPU_S + (MEM_MIB / 1024) * COST_PER_GIB_S)


def _sh(cmd, cwd, timeout):
    exports = " ".join("%s=%r" % (k, v) for k, v in ENV.items())
    t0 = time.time()
    p = subprocess.run(["bash", "-c", "export %s; %s" % (exports, cmd)], cwd=cwd, capture_output=True, text=True,
                       timeout=timeout, env={**os.environ, **ENV})
    return p.returncode, (p.stdout or "") + (p.stderr or ""), time.time() - t0


def tree_hash(top):
    """sha256 over the sorted lines '<relative path> <sha256 of the bytes>' of every file under top outside .lake."""
    rows = []
    for dp, dns, fns in os.walk(top):
        dns[:] = sorted(d for d in dns if d != ".lake")
        for fn in fns:
            p = os.path.join(dp, fn)
            rel = os.path.relpath(p, top)
            rows.append("%s %s" % (rel, hashlib.sha256(open(p, "rb").read()).hexdigest()))
    rows.sort()
    return hashlib.sha256("\n".join(rows).encode()).hexdigest(), len(rows)


@app.function(image=base, cpu=CPU, memory=MEM_MIB, timeout=TIMEOUT, retries=0, volumes={"/rec": records})
def repin(tree_tgz: bytes, commit: str, label: str) -> dict:
    t0 = time.time()
    os.makedirs("/proj", exist_ok=True)
    with tarfile.open(fileobj=io.BytesIO(tree_tgz), mode="r:gz") as tar:
        try:
            tar.extractall("/proj", filter="fully_trusted")
        except TypeError:
            tar.extractall("/proj")
    lean = "/proj/lean"
    out = {"commit": commit, "label": label, "steps": []}
    # the branch's commit carries the two edits; the container only re-resolves the manifest
    tc = Path(lean, "lean-toolchain").read_text().strip()
    lf = Path(lean, "lakefile.toml").read_text()
    assert tc == LEAN_TOOLCHAIN, "lean-toolchain is %r" % tc
    assert ('rev = "%s"' % MATHLIB_REV) in lf, "the lakefile's mathlib rev is not the joint pin"
    out["shipped_manifest_sha256"] = hashlib.sha256(Path(lean, "lake-manifest.json").read_bytes()).hexdigest()
    os.remove(Path(lean, "lake-manifest.json"))
    for name, cmd, tmo in [
        ("lean --version", "lean --version && lake --version", 120),
        ("lake update", "lake update 2>&1 | tail -30", 1200),
        ("after update", "cat lean-toolchain; grep -A2 '\"name\": \"mathlib\"' lake-manifest.json; grep -c '\"rev\"' lake-manifest.json; ls .lake/packages", 60),
        ("lake exe cache get", "lake exe cache get 2>&1 | tail -8", 1800),
        ("lake build", "/usr/bin/time -v lake build %s 2>&1" % TARGETS, 3000),
        ("axiom check", "cd /proj && python3 tools/axiom_check.py 2>&1", 900),
    ]:
        rc, log, wall = _sh(cmd, lean, tmo)
        out["steps"].append({"step": name, "rc": rc, "wall_s": round(wall, 1), "log": log[-20000:]})
        print("== %s: rc %d, %.0f s" % (name, rc, wall))
        if name == "after update":
            mb = Path(lean, "lake-manifest.json").read_bytes()
            out["manifest"] = mb.decode()
            out["manifest_sha256"] = hashlib.sha256(mb).hexdigest()
            out["toolchain_after_update"] = Path(lean, "lean-toolchain").read_text()
            out["tree_hash_after_update"], out["tree_files"] = tree_hash(lean)
            print("manifest sha256 %s; tree %s over %d files" % (out["manifest_sha256"], out["tree_hash_after_update"], out["tree_files"]))
        if name == "lake build":

            out["build_log_full"] = log
            with open("/rec/%s-repin-build-%s.log" % (time.strftime("%Y%m%d-%H%M%S", time.gmtime()), label), "w") as f:

def _ship(commit: str) -> bytes:
    with tempfile.TemporaryDirectory(prefix="specrepin-") as tmp:
        p = subprocess.run(["git", "archive", "--format=tar", commit, "lean", "tools/axiom_check.py"], cwd=ROOT,
                           capture_output=True, check=True)
        with tarfile.open(fileobj=io.BytesIO(p.stdout), mode="r:") as tar:
            tar.extractall(tmp)
        buf = io.BytesIO()
        with tarfile.open(fileobj=buf, mode="w:gz") as tar:
            tar.add(tmp, arcname=".", recursive=True)
        return buf.getvalue()


@app.local_entrypoint()
def main(commit: str = "HEAD", label: str = "s2"):
    sha = subprocess.run(["git", "rev-parse", "--short", commit], cwd=ROOT, capture_output=True, text=True, check=True).stdout.strip()
    app_id = getattr(app, "app_id", None) or "?"

    r = repin.remote(_ship(commit), sha, label)
    outdir = Path(os.environ.get("REPIN_OUT", "/tmp"))

    Path(str(outp).replace(".json", "-build.log")).write_text(r.get("build_log_full", ""))
    Path(str(outp).replace(".json", "-manifest.json")).write_text(r.get("manifest", ""))
    Path(str(outp).replace(".json", "-axioms.txt")).write_text(r.get("axiom_log_full", ""))
    for s in r["steps"]:
        print("%-20s rc %d  %7.1f s" % (s["step"], s["rc"], s["wall_s"]))
    print("errors: %d; warnings: %d; sorry lines: %d; axiom: %s; wall %.0f s; ~ $%.3f [D] at the [M] rates; app %s; json %s" % (
        len(r.get("errors", [])), len(r.get("warnings", [])), len(r.get("sorry_lines", [])), r.get("axiom_line"),
        r["wall_s"], r["price"], app_id, outp))
