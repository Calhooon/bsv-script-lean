def _price(secs):
    return secs * (CPU * COST_PER_CPU_S + (MEM_MIB / 1024) * COST_PER_GIB_S)


def _sh(cmd, cwd, timeout):
    exports = " ".join("%s=%r" % (k, v) for k, v in ENV.items())
    t0 = time.time()
    p = subprocess.run(["bash", "-c", "export %s; %s" % (exports, cmd)], cwd=cwd, capture_output=True, text=True,
                       timeout=timeout, env={**os.environ, **ENV})
    return p.returncode, (p.stdout or "") + (p.stderr or ""), time.time() - t0


@app.function(image=base, cpu=CPU, memory=MEM_MIB, timeout=TIMEOUT, retries=0, volumes={"/rec": records})
def dryrun(tree_tgz: bytes, commit: str, label: str) -> dict:
    t0 = time.time()
    os.makedirs("/proj", exist_ok=True)
    with tarfile.open(fileobj=io.BytesIO(tree_tgz), mode="r:gz") as tar:
        try:
            tar.extractall("/proj", filter="fully_trusted")
        except TypeError:
            tar.extractall("/proj")
    lean = "/proj/lean"
    out = {"commit": commit, "label": label, "steps": []}
    # the move, as road (a) would land it: the toolchain file, the lakefile's mathlib rev, the manifest re-resolved by lake
    Path(lean, "lean-toolchain").write_text(LEAN_TOOLCHAIN + "\n")
    lf = Path(lean, "lakefile.toml").read_text()
    assert 'rev = "v4.33.0"' in lf, "the lakefile's mathlib rev line moved"
    Path(lean, "lakefile.toml").write_text(lf.replace('rev = "v4.33.0"', 'rev = "%s"' % MATHLIB_REV))
    os.remove(Path(lean, "lake-manifest.json"))
    for name, cmd, tmo in [
        ("lean --version", "lean --version && lake --version", 120),
        ("lake update", "lake update 2>&1 | tail -20", 1200),
        ("manifest", "grep -A2 '\"name\": \"mathlib\"' lake-manifest.json; grep -c '\"rev\"' lake-manifest.json", 60),
        ("lake exe cache get", "lake exe cache get 2>&1 | tail -5", 1800),
        ("lake build", "/usr/bin/time -v lake build 2>&1", 3000),
        ("axiom check", "lake env lean AxiomCheck.lean 2>&1 | grep -c 'depends on axioms\\|does not depend'; lake env lean AxiomCheck.lean 2>&1 | grep -v 'propext\\|Classical.choice\\|Quot.sound\\|does not depend' | head -20", 900),
    ]:
        rc, log, wall = _sh(cmd, lean, tmo)
        out["steps"].append({"step": name, "rc": rc, "wall_s": round(wall, 1), "log": log[-20000:]})
        print("== %s: rc %d, %.0f s" % (name, rc, wall))
        if name == "lake build":
            lines = log.splitlines()
            errs = [l for l in lines if l.startswith("error:") or ": error:" in l]
            warns = [l for l in lines if l.startswith("warning:") or ": warning:" in l]
            out["errors"] = errs
            out["warnings"] = warns
            out["build_log_full"] = log
            with open("/rec/%s-toolchain-dryrun-35-build-%s.log" % (time.strftime("%Y%m%d-%H%M%S", time.gmtime()), label), "w") as f:
                f.write(log)
            print("lake build: %d error line(s), %d warning line(s)" % (len(errs), len(warns)))
        if rc != 0 and name in ("lake update", "lake exe cache get"):
            print(log[-3000:])
    out["wall_s"] = round(time.time() - t0, 1)
    out["price"] = round(_price(out["wall_s"]), 3)
    stamp = time.strftime("%Y%m%d-%H%M%S", time.gmtime())
    with open("/rec/%s-toolchain-dryrun-35-%s.log" % (stamp, label), "w") as f:
        for s in out["steps"]:
            f.write("== %s (rc %d, %.1f s)\n%s\n" % (s["step"], s["rc"], s["wall_s"], s["log"]))
    records.commit()
    out["stamp"] = stamp
    return out


def _ship(commit: str) -> bytes:
    with tempfile.TemporaryDirectory(prefix="spec35-") as tmp:
        p = subprocess.run(["git", "archive", "--format=tar", commit, "lean"], cwd=ROOT, capture_output=True, check=True)
        with tarfile.open(fileobj=io.BytesIO(p.stdout), mode="r:") as tar:
            tar.extractall(tmp)
        buf = io.BytesIO()
        with tarfile.open(fileobj=buf, mode="w:gz") as tar:
            tar.add(tmp, arcname=".", recursive=True)
        return buf.getvalue()


@app.local_entrypoint()
def main(commit: str = "HEAD", label: str = "s2"):
    sha = subprocess.run(["git", "rev-parse", "--short", commit], cwd=ROOT, capture_output=True, text=True, check=True).stdout.strip()
    app_id = getattr(app, "app_id", None) or "?"
    print("shipping lean/ at %s; app %s" % (sha, app_id))
    r = dryrun.remote(_ship(commit), sha, label)
    import json
    outp = Path(os.environ.get("SPEC35_OUT", "/tmp")) / ("dryrun35-%s.json" % r["stamp"])
    outp.write_text(json.dumps({**r, "app_id": app_id}, indent=1))
    Path(str(outp).replace(".json", "-build.log")).write_text(r.get("build_log_full", ""))
    for s in r["steps"]:
        print("%-20s rc %d  %7.1f s" % (s["step"], s["rc"], s["wall_s"]))
    print("errors: %d; warnings: %d; wall %.0f s; ≈ $%.3f [D] at the [M] rates; app %s; json %s" % (
        len(r.get("errors", [])), len(r.get("warnings", [])), r["wall_s"], r["price"], app_id, outp))
