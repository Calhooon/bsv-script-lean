"""PROPOSED ONLY. Copy to scratch after the parent's explicit launch authority.

App [LIT] spec-million-replay-s2-s10; resources [EST] 4 vCPU, 128 GiB.
Wall [LIT] 1800 s; inner wall [LIT] 1740 s; retries [LIT] 0.
Requires a verified, prepared Linux image and independently recorded binary hashes.
No image build, source fetch, node RPC, full scanner or retry is part of this app.
"""
import hashlib
import json
import os
from pathlib import Path

# Refuse even module loading until the missing artifact identifiers are supplied.
IMAGE_ID = os.environ['S2_REPLAY_IMAGE_ID']
EVAL_SHA = os.environ['S2_REPLAY_EVAL_SHA256']
DRIVER_SHA = os.environ['S2_REPLAY_DRIVER_SHA256']
assert IMAGE_ID.startswith('im-')
assert all(len(s) == 64 and all(c in '0123456789abcdef' for c in s)
           for s in (EVAL_SHA, DRIVER_SHA))
import modal

PIN = '1402ec2d139d8d636e7b6859ca2b6a054387bf5c'
WID = 'witness-million-spec-engine.resource-abort-million-seed-969309'
WITNESS_SHA = '4aef69baa868a43c2f388b5b5c5ead7c1c38b26c6eeac99d4667c9b589d31a02'
app = modal.App('spec-million-replay-s2-s10')
prepared = modal.Image.from_id(IMAGE_ID)

@app.function(image=prepared, cpu=4, memory=131072, timeout=1800,
              retries=0, single_use_containers=True)
def replay(witness: bytes):
    import shutil
    import signal
    import subprocess
    import tempfile
    import time

    source = Path('/proj')
    binary = source/'lean/.lake/build/bin/bsvs'
    driver = Path('/cache/target/release/run_diff')
    def read(args):
        return subprocess.run(args, capture_output=True, text=True,
                              check=True, timeout=15).stdout.strip()
    def git(*args):
        return read(['git','-C',str(source),*args])
    assert git('rev-parse','HEAD') == PIN
    assert not git('status','--porcelain','--','lean','eval','harness')
    assert not (source/'docs/records').exists(), 'Prepared image contains closed history'
    assert binary.is_file() and driver.is_file(), 'Prepared artifacts missing'
    assert hashlib.sha256(binary.read_bytes()).hexdigest() == EVAL_SHA
    assert hashlib.sha256(driver.read_bytes()).hexdigest() == DRIVER_SHA
    assert hashlib.sha256(witness).hexdigest() == WITNESS_SHA
    row=json.loads(witness)
    assert row['id']==WID and row['divergence']['b']=='bsv-rs@0.3.30'
    env={**os.environ,'CARGO_BUILD_JOBS':'4','BSVS_REBUILD':'0',
         'CARGO_TARGET_DIR':'/cache/target','BSV_SCRIPT_LEAN_ROOT':str(source)}
    version=subprocess.run([str(source/'eval/bsvs'),'--version'],env=env,
                           capture_output=True,text=True,timeout=30,check=True).stdout.strip()
    assert version.startswith('bsvs spec@1402ec2 ')
    root=Path(tempfile.mkdtemp(prefix='million-only-'))
    for rel in ['tools','eval','harness','corpus/witnesses']:
        (root/rel).mkdir(parents=True,exist_ok=True)
    shutil.copyfile(source/'tools/witness.sh',root/'tools/witness.sh')
    (root/'corpus/witnesses/million.jsonl').write_bytes(witness)
    (root/'eval/bsvs').write_text('#!/bin/bash\nset -eu\nexec /proj/eval/bsvs "$@"\n')
    (root/'eval/bsvs').chmod(0o755)
    log=root/'replay.log'
    samples=[]
    start=time.monotonic()
    reason=None
    p=None
    def members(pgid):
        rows=read(['ps','-eo','pgid=,stat=,rss=']).splitlines()
        return [f for line in rows if len(f:=line.split())==3
                and int(f[0])==pgid and not f[1].startswith('Z')]
    def cleanup():
        if p is None:
            return
        if members(p.pid):
            os.killpg(p.pid,signal.SIGTERM)
            end=time.monotonic()+3
            while members(p.pid) and time.monotonic()<end:
                p.poll(); time.sleep(0.05)
            if members(p.pid):
                os.killpg(p.pid,signal.SIGKILL)
        p.wait()
    def stopped(sig,frame):
        cleanup()
        raise SystemExit(128+sig)
    for sig in (signal.SIGTERM,signal.SIGINT,signal.SIGHUP):
        signal.signal(sig,stopped)
    try:
        with log.open('w') as out:
            p=subprocess.Popen(['/usr/bin/time','-v','bash',str(root/'tools/witness.sh'),WID],
                               stdout=out,stderr=subprocess.STDOUT,env=env,start_new_session=True)
            while p.poll() is None:
                elapsed=time.monotonic()-start
                rss=sum(int(f[2]) for f in members(p.pid))
                samples.append({'wall_s':round(elapsed,3),'rss_kib':rss})
                if elapsed>=1740:
                    reason='wall-bound'; break
                if rss>=120*1024*1024:
                    reason='rss-threshold'; break
                time.sleep(0.5)
            if reason:
                cleanup()
            else:
                p.wait()
    finally:
        cleanup()
    output=log.read_text(errors='replace')
    result={'source_pin':PIN,'image_id':IMAGE_ID,'eval_sha256':EVAL_SHA,
            'driver_sha256':DRIVER_SHA,'witness_sha256':WITNESS_SHA,'version':version,
            'wall_s':time.monotonic()-start,'exit':p.returncode,'stop_reason':reason,
            'green':p.returncode==0 and 'witness.sh: GREEN' in output,
            'group_gone':not members(p.pid),'samples':samples,'output':output}
    print('[M/X] replay_record '+json.dumps(result),flush=True)
    return result

@app.local_entrypoint()
def launch(witness_file: str):
    witness=Path(witness_file).read_bytes()
    assert hashlib.sha256(witness).hexdigest()==WITNESS_SHA
    call=replay.spawn(witness)
    print('[LIT] app_id='+app.app_id+' call_id='+call.object_id)
    print('Detached call submitted; book the actual meter after completion. No retry.')
